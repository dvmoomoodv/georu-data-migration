package [(${s.basePackage})].common;

import com.georu.core.rule.AuditRuleSet;
import org.springframework.stereotype.Component;

import java.util.List;

/**
 * [(${s.company})] 등록자 · 수정자 규칙. 이관 적재는 MIG-DATA, WRITER_TEST 적재는 MIG-TEST 로 표기하고
 * 테스트 적재 정리는 이 컬럼의 MIG-TEST 행을 지운다. 공통 컬럼 REG_ID · UPT_ID 기준이며 업체 컬럼이 다르면 고친다.
 * 자동 생성: georu 코드 생성기 [(${gen.version})]
 */
@Component
public class [(${s.pascal})]AuditRules implements AuditRuleSet {

    @Override
    public String migrationUser() {
        return [(${s.pascal})]ConvertUtils.MIG_USER;
    }

    @Override
    public String testUser() {
        return "MIG-TEST";
    }

    @Override
    public List<String> auditColumns() {
        return List.of("REG_ID", "UPT_ID");
    }

    @Override
    public List<String> auditSetterNames() {
        return List.of("setRegId", "setUptId");
    }
}
