      max-request-size: 55MB

    proxy-target-class: true             # Reader · Writer 의 Step 리스너 등록

    job:
      enabled: false                     # 기동 시 Job 자동 실행 안 함 (운영 UI · API 로 실행)
    jdbc:
      initialize-schema: never           # BATCH_* 는 georu.operation.schema-init 가 만든다

georu:
  customer:
    name: [(${s.company})]               # 운영 UI 이름
  datasource:
    operation:                           # 운영 저장소 (실행 이력 · 로그). 기본: 내장 H2 파일

        username: ${OPERATION_DB_USERNAME:sa}
        password: ${OPERATION_DB_PASSWORD:}
    [(${s.sourceDataSource})]:                            # 소스 (접속 정보는 프로파일 yml)
      hikari:
        driver-class-name: [(${s.sourceDriver})]
        read-only: true
        auto-commit: false
    [(${s.targetDataSource})]:                            # 타겟 (접속 정보는 프로파일 yml)
      hikari:
        driver-class-name: [(${s.targetDriver})]
        auto-commit: false
  plugins:
    [(${s.company})]:
      enabled: true                      # 업무 하나만 끄기: georu.plugins.[(${s.company})].<업무>.enabled: false

    enabled: true
    # table: mig_sequence                # 관리 테이블 (기본 mig_sequence, 타겟 DB). 예: tobe_schema.tb_sequence
    # table-name-column: TABLES          # 테이블명 열
    # column-name-column: COL            # 컬럼명 열
    # value-column: SEQ                  # 마지막 번호 열
    # key-case: UPPER                    # 관리 테이블 키 대소문자 UPPER | LOWER | AS_IS
  operation:
    schema-init: auto                    # 운영 저장소 테이블을 기동 때 만든다
  batch:
    executor:                            # 단일 실행 규칙: 엔진 기본값이 1/1 이다 (2 이상이면 기동 실패). 바꿀 일이 없으면 지워도 된다
      core-pool-size: 1
      max-pool-size: 1
